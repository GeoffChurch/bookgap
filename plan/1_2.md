```python
from enum import Enum

class OpenAIModel(Enum):
    """OpenAI model variants with their string representations."""
    GPT_4O_MINI = "gpt-4o-mini"
    GPT_4O = "gpt-4o"
    GPT_4 = "gpt-4"
```

### Step 4: Create Batch Generator Interface 🔄 **NEXT**
**File**: `bookgap/llm/generators/batch_generator.py` (new)

Abstract base class defining the batch generator contract:

```python
from abc import ABC, abstractmethod
from typing import List

from bookgap.configs.config import ExperimentConfig
from ..protocols import ProposalContext, BatchRequestData, PromptPair


class BatchProposalGenerator(ABC):
    """Abstract base class for batch proposal generators.
    
    Designed for batch processing of multiple contexts into batch API requests.
    Replaces individual request pattern with batch-first approach.
    """
    
    def __init__(self, config: ExperimentConfig):
        """Initialize generator with experiment configuration."""
        self.config = config
    
    def generate_batch_requests(self, contexts: List[ProposalContext]) -> List[BatchRequestData]:
        """Generate batch API requests from proposal contexts.
        
        Args:
            contexts: List of proposal contexts to generate requests for
            
        Returns:
            List of batch API request data structures
            
        Raises:
            ProposalGenerationError: If batch request generation fails
        """
        requests = []
        for context in contexts:
            prompts = self._build_prompt(context)  # Returns PromptPair
            request = self._create_request_from_prompt(prompts, context)
            requests.append(request)
        
        return requests
    
    @abstractmethod
    def _build_prompt(self, context: ProposalContext) -> PromptPair:
        """Build system and user prompts for a single context.
        
        Args:
            context: The proposal context
            
        Returns:
            PromptPair with system_prompt and user_prompt
        """
        ...
    
    @abstractmethod  
    def _create_request_from_prompt(self, prompts: PromptPair, context: ProposalContext) -> BatchRequestData:
        """Create batch API request from system and user prompts.
        
        Args:
            prompts: PromptPair containing system_prompt and user_prompt
            context: The original proposal context
            
        Returns:
            Batch API request data structure
        """
        ...
```

**Design Decisions**:
- **Constructor Pattern**: Takes `ExperimentConfig` directly (no legacy `AnalysisConfiguration`)
- **Batch-First**: Primary method generates list of requests, not individual requests
- **Composition**: Combines prompt building with request formatting
- **Abstract Methods**: Subclasses define provider-specific prompt and request logic

### Step 5: Implement OpenAI Batch Generator 
**File**: `bookgap/llm/generators/openai_batch.py` (new)

OpenAI-specific implementation with modernized patterns:

```python
import json
import os
from typing import Dict, Any

from openai import OpenAI
from openai.lib._pydantic import to_strict_json_schema

from bookgap.configs.config import ExperimentConfig
from bookgap.llm.providers import ProviderType
from bookgap.llm.llm_implementation import Implementation
from ..protocols import (
    ProposalContext, BatchRequestData, ChatMessage, ChatCompletionBody,
    ProposalGenerationError, create_proposal_model, PromptPair, OpenAIModel,
    Article, Book, ProposalCounts
)
from .batch_generator import BatchProposalGenerator


class OpenAIBatchGenerator(BatchProposalGenerator):
    """OpenAI Batch API generator for textbook proposals.
    
    Clean implementation using system/user message format and simple custom IDs.
    No dependency on legacy AnalysisConfiguration or CustomIdManager.
    """
    
    def __init__(self, config: ExperimentConfig):
        """Initialize OpenAI batch generator with configuration validation."""
        super().__init__(config)
        
        # Validate provider type
        if config.llm.provider_type != ProviderType.OPENAI:
            raise ProposalGenerationError(
                f"OpenAI generator requires provider_type=ProviderType.OPENAI, got '{config.llm.provider_type}'",
                context={"provider": str(config.llm.provider_type)}
            )
        
        # Validate API key
        if not os.getenv("OPENAI_API_KEY"):
            raise ProposalGenerationError(
                "OPENAI_API_KEY environment variable is required",
                context={"provider": "openai"}
            )
        
        # Store model configuration
        self.model_enum = config.llm.model_name  # Already an OpenAIModel enum
        self.model = self.model_enum.value  # Get string value for API
        self.temperature = config.llm.temperature
        self.max_tokens = config.llm.max_tokens
        self.use_structured = config.llm.use_structured_output
        
    def _build_prompt(self, context: ProposalContext) -> PromptPair:
        """Build system and user prompts separately for native message format.
        
        Returns:
            PromptPair with system_prompt and user_prompt
        """
        system_prompt = self._build_system_prompt()
        user_prompt = self._build_user_prompt(context)
        return PromptPair(system_prompt=system_prompt, user_prompt=user_prompt)
    
    def _create_request_from_prompt(self, prompts: PromptPair, context: ProposalContext) -> BatchRequestData:
        """Create OpenAI batch API request from system/user prompts."""
        
        messages = [
            ChatMessage(role="system", content=prompts.system_prompt),
            ChatMessage(role="user", content=prompts.user_prompt)
        ]
        
        # Build request body
        body_params = {
            "model": self.model,
            "messages": messages,
            "max_tokens": self.max_tokens,
        }
        
        # Add temperature if model supports it
        if self.model_enum.supports_temperature():
            body_params["temperature"] = self.temperature
        
        # Add structured output schema if enabled
        if self.use_structured:
            proposal_model = create_proposal_model(
                chapters_min=self.config.llm.chapters_min,
                chapters_max=self.config.llm.chapters_max
            )
            body_params["response_format"] = {
                "type": "json_schema",
                "json_schema": {
                    "name": "textbook_proposal",
                    "schema": to_strict_json_schema(proposal_model),
                    "strict": True
                }
            }
        
        body = ChatCompletionBody(**body_params)
        
        # Simple custom ID (no embedded configuration)
        custom_id = f"gap_{context.gap_idx}_topic_{context.topic_id}"
        
        return BatchRequestData(
            custom_id=custom_id,
            method="POST",
            url="/v1/chat/completions",
            body=body
        )
    
    def _build_system_prompt(self) -> str:
        """Build system prompt with role definition."""
        return "You are a publishing expert and academic researcher."
    
    def _build_user_prompt(self, context: ProposalContext) -> str:
        """Build user prompt from context with clean formatting."""
        prompt_parts = []
        
        # Introduction
        prompt_parts.append(
            "Based on the research articles"
        )
        if context.covering_books:
            prompt_parts.append(" and related books")
        prompt_parts.append(
            " provided below, please propose a title, abstract, and table of contents "
            "for a textbook that would prepare students to understand research in this area.\n\n"
        )
        
        # Sample articles section
        prompt_parts.append("## Sample Articles\n\n")
        for i, article in enumerate(context.sampled_articles, 1):
            prompt_parts.append(f"**Article {i}**: {article.display_name}\n")
            prompt_parts.append(f"\t*Abstract*: {article.abstract}\n\n")
        
        # Related books section (if any)
        if context.covering_books:
            prompt_parts.append("## Related Existing Books\n\n")
            for i, book in enumerate(context.covering_books, 1):
                prompt_parts.append(f"**Book {i}**: {book.display_name}\n")
                prompt_parts.append(f"\t*Abstract*: {book.abstract}\n\n")
        
        # Guidelines
        prompt_parts.append("## Guidelines\n\n")
        min_chapters = self.config.llm.chapters_min
        max_chapters = self.config.llm.chapters_max
        prompt_parts.append(f"- Include {min_chapters}-{max_chapters} chapters.\n")
        prompt_parts.append("- Build up systematically from basic to advanced concepts.\n")
        prompt_parts.append("- Each chapter should have a clear title and description of concepts covered.\n")
        
        return "".join(prompt_parts)
```

**Key Modernizations:**
- **Native Message Format**: Returns `PromptPair` dataclass from `_build_prompt()` 
- **Simple Custom IDs**: No embedded configuration, clean format
- **Direct Config Usage**: No `AnalysisConfiguration` bridge needed
- **Clean Prompt Building**: Extracted and simplified from existing patterns

```python
from typing import List

from bookgap.configs.config import ExperimentConfig
from bookgap.llm.providers import ProviderType  
from ..protocols import ProposalContext, BatchRequestData
from .openai_batch import OpenAIBatchGenerator


def generate_batch_requests(contexts: List[ProposalContext], config: ExperimentConfig) -> List[BatchRequestData]:
    """Generate batch API requests from proposal contexts.
    
    Uses Hydra's _target_ pattern to instantiate the appropriate generator for the provider.
    
    Args:
        contexts: List of proposal contexts
        config: Experiment configuration
        
    Returns:
        List of batch API request data structures
        
    Raises:
        ProposalGenerationError: If generator instantiation or generation fails
    """
    from hydra.utils import instantiate
    
    # Instantiate generator using Hydra's _target_ pattern
    generator = instantiate(config.llm, config=config)
    
    return generator.generate_batch_requests(contexts)
```

**Hydra `_target_` Benefits**:
- **Configuration-Driven**: Generator class specified in config, not hardcoded business logic
- **Runtime Flexibility**: Can swap generators via config overrides (`python script.py llm._target_=my_app.ClaudeGenerator`)
- **Easy Provider Addition**: Adding Claude/Gemini just requires new config presets, zero code changes
- **Native Hydra Integration**: Leverages built-in instantiation system designed for this exact use case
