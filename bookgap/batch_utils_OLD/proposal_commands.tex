% Enhanced LaTeX commands for textbook proposals with premium styling
% Include this file in your document preamble with: \input{proposal_commands_premium}
% Requires: enumitem, ragged2e, xcolor, fancyhdr packages

% Define colors for a professional look
\definecolor{proposalblue}{RGB}{25,25,112}     % MidnightBlue
\definecolor{proposalgray}{RGB}{105,105,105}   % DimGray
\definecolor{proposallight}{RGB}{248,248,255}  % GhostWhite

% Command for proposal title - large, centered, colored, with decorative rule
\newcommand{\proposalTitle}[1]{%
    \begin{center}%
        \vspace{1em}%
        {\color{proposalblue}\Large\textbf{#1}}\\[0.5em]%
        {\color{proposalgray}\rule{0.6\textwidth}{0.5pt}}%
    \end{center}%
    \vspace{1em}%
}

% Command for proposal abstract - centered, in a subtle box
\newcommand{\proposalAbstract}[1]{%
    \begin{center}%
        \fcolorbox{proposalgray}{proposallight}{%
            \begin{minipage}{0.85\textwidth}%
                \centering%
                {\color{proposalblue}\textbf{Abstract}}\\[0.75em]%
                \justifying%
                #1%
                \vspace{0.5em}%
            \end{minipage}%
        }%
    \end{center}%
    \vspace{1.5em}%
}

% Enhanced chapters command with professional styling
\newcommand{\proposalChapters}[1]{%
    \begin{center}%
        {\color{proposalblue}\textbf{Table of Contents}}\\[0.5em]%
        {\color{proposalgray}\rule{0.4\textwidth}{0.3pt}}%
    \end{center}%
    \vspace{1em}%
    \begin{enumerate}[%
        leftmargin=2.5em,%
        itemsep=0.75em,%
        label=\textbf{\color{proposalblue}\arabic*.},%
        font=\color{proposalblue}%
    ]%
        #1%
    \end{enumerate}%
    \vspace{1em}%
}

% Enhanced chapter item with sophisticated typography
\newcommand{\chapterItem}[2]{%
    \item \textbf{#1} \\%
    \hspace{1em}{\color{proposalgray}\textit{#2}}%
}

% Alternative simpler but still elegant version
\newcommand{\proposalTitleSimple}[1]{%
    \begin{center}%
        {\LARGE\textsc{#1}}%
    \end{center}%
    \vspace{1em}%
}

\newcommand{\proposalAbstractSimple}[1]{%
    \begin{center}%
        \begin{minipage}{0.8\textwidth}%
            \centering%
            \textbf{Abstract}\\[0.5em]%
            \justifying%
            \textit{#1}%
        \end{minipage}%
    \end{center}%
    \vspace{1.5em}%
}
